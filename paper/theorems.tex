\section{Results}\label{sec:results}
\begin{theorem}[Multi-round factorization; Lean \texttt{runRounds\_congr}]
If every round transition factors through an observation function $A$, equal observations in every round imply equal final states.
\end{theorem}
\begin{theorem}[Blind impossibility; Lean \texttt{blind\_consensus\_impossible}]
For every $n\ge2$, no $L_0^{\mathrm{blind}}$ algorithm solves binary consensus, even with $f=0$.
\end{theorem}
\paragraph{Proof sketch.} With $f=0$ every process sees only the same fixed count vector, so its decision depends only on its own input. Validity on uniform executions forces different outputs for different inputs, while a mixed execution requires agreement.
\begin{theorem}[Crusader agreement at $L_0$]
For $n>3f$, the one-round threshold rule \textquotedblleft adopt $v$ iff at least $n-f$ copies of $v$ arrive; otherwise $\bot$\textquotedblright\ solves crusader agreement.
\end{theorem}
\begin{theorem}[Connected consensus $R=2$ at $L_0$]
For $n>3f$, the two-round graded threshold protocol in \texttt{CountConnectedConsensus} satisfies spider-graph agreement and validity.
\end{theorem}
\begin{theorem}[One-dimensional approximate agreement at $L_0$]
For $n>3f$, trimmed midpoint iterations reduce the honest diameter geometrically.
\end{theorem}
\begin{theorem}[Multivalued Byzantine agreement at $L_1$]
For $n>3f$, the phase-king implementation uses sender identity only for the fixed king and runs for $3(f+1)$ rounds.
\end{theorem}
\begin{lemma}[Anonymous counts cannot name the king; Lean \texttt{king\_not\_count\_definable}]
No function of anonymous counts can return the value sent by a fixed sender.
\end{lemma}
\paragraph{Main open separation.} The software treats $L_0<L_1$ for Byzantine agreement as a candidate. The homonyms literature must be checked against the exact anonymous-count model, or a direct symmetry impossibility must be supplied before promotion to theorem.
\paragraph{Open.} Connected consensus for $R\ge3$, multidimensional approximate agreement at $L_0$, and an $L_1/L_2$ separation under bounded certificates remain open.
